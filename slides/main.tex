\begin{frame}{Problem Statement}
    \small
    Boxes arrive on a belt; a robot picks a consecutive group, places it on the pallet, and repeats.

    \vspace{0.25cm}
    \begin{columns}[T,totalwidth=\textwidth]
        \begin{column}{0.44\textwidth}
            \textbf{Given}
            \begin{itemize}
                \item $n$ boxes of size $x \times y$, arriving in order
                \item A pallet grid of $W$ columns $\times$ $R$ rows
                \item Gripper size $S$
            \end{itemize}

            \vspace{0.2cm}
            \textbf{Decide, for each pick}
            \begin{itemize}
                \item how many consecutive boxes to grab
                \item where on the pallet to drop them
                \item which plate opens at release
            \end{itemize}
        \end{column}
        \begin{column}{0.52\textwidth}
            \textbf{Subject to}
            \begin{itemize}
                \item picks are contiguous and cover the belt with no gaps
                \item each pick spans at most $\lfloor(S+x)/x\rfloor$ boxes
                \item all boxes on the same pick go to the same row
                \item no two picks land on the same cell
                \item the opening plate faces only free space at the moment of release
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \textbf{Goal:} find \emph{any} feasible assignment: no objective to minimize.
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{Parameters and Derived Constants}
    \textbf{Input parameters}
    \begin{lstlisting}
par int: n_boxes;       % total number of boxes
par int: boxes_per_row; % boxes in each pallet row
par int: pallet_rows;   % number of rows on the pallet
par int: boxes_x_dim;   % box dimension parallel to the gripper
par int: grip_size;     % gripper size S
    \end{lstlisting}

    \vspace{0.3cm}
    \textbf{Derived constants}
    \begin{lstlisting}
par int: max_pkble_boxes =
  (grip_size + boxes_x_dim) div boxes_x_dim;

par int: chunks_per_row =
  (boxes_per_row + max_pkble_boxes - 1) div max_pkble_boxes;

par int: n_chunks = pallet_rows * chunks_per_row;
    \end{lstlisting}
\end{frame}

%------------------------------------------------

\begin{frame}{Chunking: Encoding Alternatives}
    \small
    Several encodings are possible for representing which boxes belong to each chunk.

    \vspace{0.2cm}
    \begin{tabular}{p{0.24\textwidth} p{0.35\textwidth} p{0.32\textwidth}}
        \toprule
        \textbf{Encoding} & \textbf{Description} & \textbf{Limitation} \\
        \midrule
        Enumerated
        & Store the list of box ids in each chunk, padded to the maximum length.
        & Membership is explicit, but many variables and ordering constraints are needed. \\
        \midrule
        Binary assignment
        & One Boolean variable per (chunk, box) pair.
        & Very large; contiguity must be enforced with additional constraints. \\
        \midrule
        Interval
        & Store only the \emph{start index} and \emph{length} of each chunk.
        & \textbf{Adopted}: compact, and contiguity is implicit by construction. \\
        \bottomrule
    \end{tabular}
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{Chunking: Variables}
    Each chunk is represented as an interval: a start index and a length.

    \vspace{0.4cm}
    \begin{lstlisting}
array[1..n_chunks] of var 1..n_boxes:         chunk_start;
array[1..n_chunks] of var 1..max_pkble_boxes: chunk_len;
    \end{lstlisting}

    \vspace{0.4cm}
    \small
    Contiguity is implicit: once start and length are fixed, the boxes in the chunk
    are uniquely determined: no explicit membership variables needed.
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{Chunking: Constraints}
    \textbf{Start at box 1}
    \begin{lstlisting}
constraint chunk_start[1] = 1;
    \end{lstlisting}

    \vspace{0.25cm}
    \textbf{No gaps or overlaps}
    \begin{lstlisting}
constraint forall(n in 1..n_chunks - 1)(
  chunk_start[n + 1] = chunk_start[n] + chunk_len[n]
);
    \end{lstlisting}

    \vspace{0.25cm}
    \textbf{Full coverage}
    \begin{lstlisting}
constraint chunk_start[n_chunks] + chunk_len[n_chunks] - 1 = n_boxes;
    \end{lstlisting}
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{Placement: Variables}
    Each chunk is assigned a row and a starting column on the pallet.

    \vspace{0.4cm}
    \begin{lstlisting}
array[1..n_chunks] of var 1..pallet_rows:   y_row;
array[1..n_chunks] of var 1..boxes_per_row: x_start;
    \end{lstlisting}
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{Placement: Constraints}
    \textbf{Row fit}
    \begin{lstlisting}
constraint forall(n in 1..n_chunks)(
  x_start[n] + chunk_len[n] - 1 <= boxes_per_row
);
    \end{lstlisting}

    \vspace{0.25cm}
    \textbf{No overlap}
    \begin{lstlisting}
constraint diffn(x_start, y_row, chunk_len, [1 | g in 1..n_chunks]);
    \end{lstlisting}

    \vspace{0.25cm}
    \textbf{Exact row coverage}
    \begin{lstlisting}
constraint global_cardinality(
  y_row,
  [r | r in 1..pallet_rows],
  [chunks_per_row | r in 1..pallet_rows]
);
    \end{lstlisting}
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{Plate Variable and Target Row}
    A binary variable encodes which side opens: $0$ = upward (row $r-1$),\ $1$ = downward (row $r+1$).

    \vspace{0.3cm}
    \begin{lstlisting}
array[1..n_chunks] of var 0..1: plate;
    \end{lstlisting}

    \vspace{0.4cm}
    An auxiliary variable computes the swept row. Boundary values $0$ and
    $\texttt{pallet\_rows}+1$ represent a free pallet edge.

    \vspace{0.3cm}
    \begin{lstlisting}
array[1..n_chunks] of var 0..pallet_rows+1: target;
constraint forall(g in 1..n_chunks)(
  target[g] = y_row[g] - 1 + 2*plate[g]
);
    \end{lstlisting}
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{No-Collision Constraint}
    Chunks are released in belt order, so any earlier chunk is already on the pallet.
    The constraint forbids an earlier chunk from sitting in the swept row and overlapping along $x$.

    \vspace{0.3cm}
    \begin{lstlisting}
constraint forall(curr_chunk in 2..n_chunks,
                  pred_chunk in 1..curr_chunk-1)(
  not(
       y_row[pred_chunk] = target[curr_chunk]
    /\ x_start[pred_chunk] < x_start[curr_chunk] + chunk_len[curr_chunk]
    /\ x_start[curr_chunk] < x_start[pred_chunk] + chunk_len[pred_chunk]
  )
);
    \end{lstlisting}
\end{frame}

%------------------------------------------------

\begin{frame}[fragile]{Search Strategy}
    Fill top-to-bottom, left-to-right, longest pick first.
    The row above is always full when a new chunk is placed,
    so the plate always opens safely downward — near-zero backtracking.

    \vspace{0.25cm}
    The four per-chunk decisions are packed into one \texttt{branch} array.
    Variables where we want the \emph{maximum} are stored as their complement,
    so a single \texttt{indomain\_min} covers all four.

    \vspace{0.2cm}
    \begin{lstlisting}
par int: branch_ub = max([pallet_rows, boxes_per_row, max_pkble_boxes]);
array[1..4*n_chunks] of var 0..branch_ub: branch;

constraint forall(i in 1..n_chunks)(
  branch[4*i-3] = y_row[i]                        /\  % min -> topmost row
  branch[4*i-2] = max_pkble_boxes - chunk_len[i]  /\  % min -> longest pick
  branch[4*i-1] = x_start[i]                      /\  % min -> leftmost column
  branch[4*i]   = 1 - plate[i]                        % min -> plate = 1 (down)
);

solve :: int_search(branch, input_order, indomain_min) satisfy;
    \end{lstlisting}
\end{frame}

\begin{frame}{Benchmark Setup}
    \small
    The benchmark tests every (pallet, box size) combination across four solvers and three gripper sizes.

    \vspace{0.2cm}
    \begin{columns}[T,totalwidth=\textwidth]
        \begin{column}{0.50\textwidth}
            \begin{table}
                \centering
                \small
                \begin{tabular}{lr}
                    \toprule
                    \textbf{Pallet preset} & \textbf{Dimensions (mm)} \\
                    \midrule
                    Europallet      & $1200 \times 800$  \\
                    Industrie       & $1200 \times 1000$ \\
                    US 40$\times$48 & $1219 \times 1016$ \\
                    Half-pallet     & $800  \times 600$  \\
                    Australian      & $1165 \times 1165$ \\
                    \bottomrule
                \end{tabular}
            \end{table}
        \end{column}
        \begin{column}{0.46\textwidth}
            \begin{itemize}
                \item Box sizes: $50 \times 50$ to $200 \times 200$ mm.
                \item Solvers: \texttt{gecode}, \texttt{cp-sat}, \texttt{chuffed}, \texttt{highs}.
                \item Gripper sizes: $S \in \{1,2,3\} \cdot \texttt{boxes\_x\_dim}$.
                \item Timeout: 300 seconds per run.
            \end{itemize}
            \vspace{0.1cm}
            \small
            \texttt{gecode}/\texttt{chuffed} use CP propagation,
            \texttt{cp-sat} uses a SAT-based representation,
            \texttt{highs} uses linear programming.
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Scaling by Gripper Size}
    \small
    \begin{columns}[T,totalwidth=\textwidth]
        \begin{column}{0.36\textwidth}
            Larger grippers reduce the number of chunks that must be scheduled:
            with $k=3$ there are roughly half as many chunks as with $k=1$.

            \vspace{0.25cm}
            This is reflected in the runtime curves. \texttt{gecode} and \texttt{chuffed}
            stay in the lowest range for all multipliers.
            \texttt{cp-sat} is slower but remains well below the timeout.
            \texttt{highs} reaches the timeout on the largest instances.
